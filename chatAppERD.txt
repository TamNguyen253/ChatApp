User [icon: user, color: blue] {
  id string pk
  username string @unique [required]
  hashedPassword string [required]
  displayName string [required]
  email string @unique
  avatarUrl string
  avatarId string
  bio string
  phone string
  createdAt datetime
  updatedAt datetime
}

FriendRequest [icon: users, color: orange] {
  id string pk
  from User [ref, required]
  to User [ref, required]
  message string
  createdAt datetime
  updatedAt datetime
}

Friend [icon: users, color: yellow] {
  id string pk
  userA User [ref, required]
  userB User [ref, required]
  createdAt datetime
  updatedAt datetime
}

Conversation [color: purple] {
  id string pk
  participants [User] [ref, required]
  type enum('direct', 'group') [required]
  group {name: string, createdBy: User [ref]}
  lastMessage {content: string, createdAt: datetime, sender: User [ref]}
  seenBy [User] [ref]
  unreadCounts Map(User, number)
  createdAt datetime
  updatedAt datetime
}

Message [color: green, icon: message-circle] {
  id string pk
  senderId User [ref, required]
  conversationId Conversation [ref, required]
  content string
  imgUrl string
  createdAt datetime
  updatedAt datetime
}

User.id < FriendRequest.from
User.id < FriendRequest.to

User.id < Friend.userA
User.id < Friend.userB

User.id < Message.senderId
Conversation.id < Message.conversationId
